function mean(a){ var s=0; for(var i=0;i<a.length;i++)s+=a[i]; return s/a.length; }
function sd(a){ var m=mean(a),s=0; for(var i=0;i<a.length;i++)s+=(a[i]-m)*(a[i]-m); return Math.sqrt(s/a.length); }
function variance(a){ var m=mean(a),s=0; for(var i=0;i<a.length;i++)s+=(a[i]-m)*(a[i]-m); return s/a.length; }
function mulberry32(a){ return function(){ a|=0; a=a+0x6D2B79F5|0; var t=Math.imul(a^a>>>15,1|a); t=t+Math.imul(t^t>>>7,61|t)^t; return ((t^t>>>14)>>>0)/4294967296; }; }
var RNG = mulberry32(42);
function solve(A,b){ var n=b.length, M=A.map(function(r,i){ return r.concat([b[i]]); });
  for(var i=0;i<n;i++){ var p=i; for(var k=i+1;k<n;k++){ if(Math.abs(M[k][i])>Math.abs(M[p][i]))p=k; }
    var tp=M[i]; M[i]=M[p]; M[p]=tp; var piv=M[i][i]||1e-12;
    for(var j=i;j<=n;j++)M[i][j]/=piv;
    for(var k=0;k<n;k++){ if(k!==i){ var f=M[k][i]; for(var j2=i;j2<=n;j2++)M[k][j2]-=f*M[i][j2]; } } }
  return M.map(function(r){ return r[n]; }); }

/* ══════════════ 3. CONJECTURA DE COLLATZ ══════════════ */
function collatz(n){ var s=[n]; while(n!==1){ n = (n%2===0) ? n/2 : 3*n+1; s.push(n); } return s; }

/* ══════════════ 4. RANDOM FOREST (árvores de regressão + bagging) ══════════════ */
function buildTree(X,y,depth,maxDepth,nFeat,imp){
  if(depth>=maxDepth||X.length<4){ return {v:mean(y)}; }
  var idx=[],nF=X[0].length; for(var i=0;i<nF;i++)idx.push(i);
  for(var i2=idx.length-1;i2>0;i2--){ var j=Math.floor(RNG()*(i2+1)); var tmp=idx[i2]; idx[i2]=idx[j]; idx[j]=tmp; }
  var sub=idx.slice(0,nFeat), parentLoss=variance(y)*y.length, best=null;
  for(var s=0;s<sub.length;s++){ var f=sub[s], vals=[];
    for(var k=0;k<X.length;k++){ if(vals.indexOf(X[k][f])<0)vals.push(X[k][f]); }
    vals.sort(function(a,b){ return a-b; });
    for(var q=0;q<vals.length-1;q++){ var thr=(vals[q]+vals[q+1])/2, Ly=[],Ry=[];
      for(var k2=0;k2<X.length;k2++){ (X[k2][f]<=thr?Ly:Ry).push(y[k2]); }
      if(Ly.length<2||Ry.length<2)continue;
      var loss=variance(Ly)*Ly.length+variance(Ry)*Ry.length;
      if(!best||loss<best.loss)best={f:f,thr:thr,loss:loss}; } }
  if(!best)return {v:mean(y)};
  imp[best.f]+=(parentLoss-best.loss);
  var LX=[],Ly=[],RX=[],Ry=[];
  for(var k3=0;k3<X.length;k3++){ if(X[k3][best.f]<=best.thr){ LX.push(X[k3]);Ly.push(y[k3]); } else { RX.push(X[k3]);Ry.push(y[k3]); } }
  return {f:best.f,thr:best.thr,l:buildTree(LX,Ly,depth+1,maxDepth,nFeat,imp),r:buildTree(RX,Ry,depth+1,maxDepth,nFeat,imp)};
}
function treePredict(t,r){ return (t.v!==undefined) ? t.v : (r[t.f]<=t.thr ? treePredict(t.l,r) : treePredict(t.r,r)); }
function trainRF(X,y,nTrees,maxDepth){
  var nFeat=Math.max(2,Math.round(Math.sqrt(X[0].length))), imp=new Array(X[0].length).fill(0), trees=[];
  for(var t=0;t<nTrees;t++){ var bx=[],by=[];
    for(var i=0;i<X.length;i++){ var k=Math.floor(RNG()*X.length); bx.push(X[k]); by.push(y[k]); }
    trees.push(buildTree(bx,by,0,maxDepth,nFeat,imp)); }
  var tot=imp.reduce(function(a,b){ return a+b; },0)||1;
  return { predict:function(r){ return mean(trees.map(function(t){ return treePredict(t,r); })); },
           importance:imp.map(function(v){ return v/tot; }) };
}

function r2(yt,yp){ var m=mean(yt),ss=0,st=0; for(var i=0;i<yt.length;i++){ ss+=(yt[i]-yp[i])*(yt[i]-yp[i]); st+=(yt[i]-m)*(yt[i]-m); } return st===0?0:1-ss/st; }
function rmse(yt,yp){ var s=0; for(var i=0;i<yt.length;i++)s+=(yt[i]-yp[i])*(yt[i]-yp[i]); return Math.sqrt(s/yt.length); }
function mae(yt,yp){ var s=0; for(var i=0;i<yt.length;i++)s+=Math.abs(yt[i]-yp[i]); return s/yt.length; }

function runSimulation(){
  var L=T[LANG];
  var colN=parseInt(document.getElementById('sel-n').value,10);
  var polMult=parseFloat(document.getElementById('sel-policy').value);
  var seqFull=collatz(colN), seqMax=Math.max.apply(null,seqFull);
  var nSteps=seqFull.length-1;

  /* features: 7 económicas + 2 Collatz */
  function feats(r){ var v=[r.barril,r.prod,r.cambio,r.inflacao,r.expPet,r.ide,r.pol]; return v; }
  var Xtrain=[], y=PIB.slice();
  DATA.forEach(function(r,i){ var cv=seqFull[i%seqFull.length];
    Xtrain.push(feats(r).concat([cv/seqMax, cv%2===0?1:-1])); });

  /* split temporal 80/20 */
  var nCut=Math.floor(Xtrain.length*0.8);
  var Xtr=Xtrain.slice(0,nCut), Xte=Xtrain.slice(nCut);
  var ytr=y.slice(0,nCut), yte=y.slice(nCut);

  /* padronização (média/desvio do treino) */
  var nF=Xtr[0].length, mu=[], sg=[];
  for(var j=0;j<nF;j++){ var col=Xtr.map(function(r){ return r[j]; }); mu.push(mean(col)); sg.push(sd(col)||1); }
  function scaleRow(r){ return r.map(function(v,j){ return (v-mu[j])/sg[j]; }); }
  var XtrS=Xtr.map(scaleRow), XteS=Xte.map(scaleRow);

  /* Regressão Linear (OLS) sobre features padronizadas */
  function withBias(X){ return X.map(function(r){ return [1].concat(r); }); }
  var Xb=withBias(XtrS), XtX=[], Xty=[];
  for(var a=0;a<nF+1;a++){ Xty.push(0); var row=[]; for(var b=0;b<nF+1;b++)row.push(0); XtX.push(row); }
  for(var i=0;i<Xb.length;i++){ for(var a2=0;a2<nF+1;a2++){ Xty[a2]+=Xb[i][a2]*ytr[i];
    for(var b2=0;b2<nF+1;b2++) XtX[a2][b2]+=Xb[i][a2]*Xb[i][b2]; } }
  var beta=solve(XtX,Xty);
  function lrPredict(r){ var s=beta[0], rs=scaleRow(r); for(var j=0;j<nF;j++)s+=beta[j+1]*rs[j]; return s; }
  var lrTr=Xtr.map(lrPredict), lrTe=Xte.map(lrPredict);

  /* Random Forest */
  var rf=trainRF(Xtr,ytr,40,5);
  var rfTr=Xtr.map(function(r){ return rf.predict(r); }), rfTe=Xte.map(function(r){ return rf.predict(r); });

  /* Ensemble = média LR + RF */
  var ensTr=lrTr.map(function(v,i){ return (v+rfTr[i])/2; });
  var ensTe=lrTe.map(function(v,i){ return (v+rfTe[i])/2; });

  /* KPIs */
  setText('kpi-gdp', PIB[PIB.length-1].toFixed(1));
  setText('kpi-barrel', '$'+BARREL[BARREL.length-1].toFixed(1));
  setText('kpi-lr', r2(yte,lrTe).toFixed(4));
  setText('kpi-rf', r2(yte,rfTe).toFixed(4));

  /* Tabela de dados */
  document.getElementById('data-table-body').innerHTML = DATA.map(function(r){
    return '<tr><td class="val-gold">'+r.ano+'</td><td class="val-primary">'+r.pib+'</td><td>'+r.barril+'</td><td>'+r.prod.toLocaleString()+'</td><td>'+r.cambio.toLocaleString()+'</td><td>'+r.inflacao+'</td><td>'+r.expPet+'</td></tr>';
  }).join('');

  /* Tabela de métricas */
  var rows=[['Linear Regression',lrTr,lrTe],['Random Forest',rfTr,rfTe],['Ensemble (LR+RF)',ensTr,ensTe]];
  document.getElementById('metrics-body').innerHTML = rows.map(function(m){
    return '<tr><td class="val-primary">'+m[0]+'</td><td>'+r2(ytr,m[1]).toFixed(4)+'</td><td>'+r2(yte,m[2]).toFixed(4)+'</td><td>'+rmse(yte,m[2]).toFixed(2)+'</td><td>'+mae(yte,m[2]).toFixed(2)+'</td></tr>';
  }).join('');

  /* Importância das features */
  var labels=FEAT_LABELS[LANG];
  var imps=rf.importance.map(function(v,i){ return {name:labels[i],val:v}; }).sort(function(a,b){ return b.val-a.val; }).slice(0,9);
  document.getElementById('importance-bars').innerHTML = imps.map(function(f){
    return '<div class="bar-item"><span class="bar-name">'+f.name+'</span><div class="bar-track"><div class="bar-fill" style="width:'+(f.val*100).toFixed(1)+'%"></div></div><span class="bar-val">'+(f.val*100).toFixed(1)+'%</span></div>';
  }).join('');

  /* Collatz: estatísticas + gráfico */
  setText('col-steps', nSteps);
  setText('col-peak', seqMax.toLocaleString());
  setText('col-mean', mean(seqFull).toFixed(1));

  makeChart('collatzChart',{ type:'bar',
    data:{ labels:seqFull.map(function(_,i){ return i; }), datasets:[{ label:'Collatz('+colN+')', data:seqFull, backgroundColor:cColors, borderRadius:3 }] },
    options:{ responsive:true, maintainAspectRatio:false, plugins:{ legend:{ display:false } },
      scales:{ x:{ ticks:{ color:'#888' } }, y:{ ticks:{ color:'#888' }, grid:{ color:'#222' } } } } });

  /* Gráficos de dados */
  makeChart('gdpChart',{ type:'line',
    data:{ labels:YEARS, datasets:[{ label:'PIB (Bn USD)', data:PIB, borderColor:'#cc2222', backgroundColor:'rgba(204,34,34,0.15)', borderWidth:2, tension:0.35, fill:true, pointRadius:2 }] },
    options:{ responsive:true, maintainAspectRatio:false, plugins:{ legend:{ labels:{ color:'#888' } } },
      scales:{ x:{ ticks:{ color:'#888', maxTicksLimit:8 }, grid:{ color:'#1a1a1a' } }, y:{ ticks:{ color:'#888' }, grid:{ color:'#222' } } } } });
  makeChart('oilChart',{ type:'line',
    data:{ labels:YEARS, datasets:[
      { label:T[LANG].th_barrel, data:BARREL, borderColor:'#e67e22', borderWidth:2, tension:0.35, pointRadius:2, yAxisID:'y' },
      { label:T[LANG].th_exp, data:EXP, borderColor:'#f5c518', borderWidth:2, tension:0.35, pointRadius:2, yAxisID:'y1' }] },
    options:{ responsive:true, maintainAspectRatio:false,
      plugins:{ legend:{ labels:{ color:'#888' } } },
      scales:{ y:{ position:'left', ticks:{ color:'#888' }, grid:{ color:'#222' } },
               y1:{ position:'right', ticks:{ color:'#888' }, grid:{ drawOnChartArea:false } },
               x:{ ticks:{ color:'#888', maxTicksLimit:8 }, grid:{ color:'#1a1a1a' } } } } });

  var FC_YEARS=[]; for(var yy=2027;yy<=2033;yy++)FC_YEARS.push(yy);
  var ultimo=DATA[DATA.length-1];

    var proj={}; ECON.forEach(function(k){ proj[k]= (k==='pol') ? 0.5 : (ultimo[k]+trends[k]*ahead); });
    var idx=(yv-2027)%seqFull.length, cVal=seqFull[idx], cNorm=cVal/seqMax;

    var lrV=lrPredict(row), rfV=rf.predict(row), ensV=(lrV+rfV)/2*polMult;
    fcLR.push(lrV); fcRF.push(rfV); fcENS.push(ensV);
    var phase=cNorm>0.7?'peak':cNorm<0.2?'trough':'normal';

  }).join('');
  var growth=(fcENS[fcENS.length-1]-fcENS[0])/fcENS[0]*100;
  document.getElementById('fc-summary').innerHTML = T[LANG].summary(colN,growth,fcENS[0].toFixed(1),fcENS[fcENS.length-1].toFixed(1));

  makeChart('fcChart',{ type:'line',
